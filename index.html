function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var data = JSON.parse(e.postData.contents); var action = data.action || "save"; var items = data.items || []; var ss = SpreadsheetApp.getActiveSpreadsheet(); var headers = ["id", "step", "name", "subRecipe", "category", "itemType", "grams", "ratePerKg", "obtainedCount", "purchasePrice", "usedCount", "checked"]; // 1. केवल बैकअप बटन दबाने पर 'Latest_Backup' शीट अपडेट होगी if (action === "backup") { var backupSheet = ss.getSheetByName("Latest_Backup"); if (!backupSheet) { backupSheet = ss.insertSheet("Latest_Backup"); } backupSheet.clear(); var timestamp = new Date().toLocaleString("hi-IN", { timeZone: "Asia/Kolkata" }); backupSheet.appendRow(["--- आख़िरी सुरक्षित बैकअप समय (Last Backup Timestamp): ---", timestamp]); backupSheet.appendRow(headers); items.forEach(function(item) { backupSheet.appendRow([ item.id || "", item.step || "", item.name || "", item.subRecipe || "", item.category || "", item.itemType || "Grocery", item.grams || 0, item.ratePerKg || 0, item.obtainedCount || 0, item.purchasePrice || 0, item.usedCount || 0, item.checked ? "true" : "false" ]); }); return ContentService.createTextOutput(JSON.stringify({status: "ok", timestamp: timestamp})).setMimeType(ContentService.MimeType.JSON); } // 2. सामान्य 'सेव करें' या 'रिस्टोर' के बाद मुख्य 'Ingredients' शीट अपडेट होगी var sheet = ss.getSheetByName("Ingredients"); if (!sheet) { sheet = ss.insertSheet("Ingredients"); } sheet.clear(); sheet.appendRow(headers); items.forEach(function(item) { sheet.appendRow([ item.id || "", item.step || "", item.name || "", item.subRecipe || "", item.category || "", item.itemType || "Grocery", item.grams || 0, item.ratePerKg || 0, item.obtainedCount || 0, item.purchasePrice || 0, item.usedCount || 0, item.checked ? "true" : "false" ]); }); return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : "load"; // रिस्टोर करने पर 'Latest_Backup' शीट से डेटा और समय उठाएगा if (action === "restore") { var backupSheet = ss.getSheetByName("Latest_Backup"); if (!backupSheet) { return ContentService.createTextOutput(JSON.stringify({status: "empty", message: "कोई बैकअप शीट नहीं मिली।"})).setMimeType(ContentService.MimeType.JSON); } var bRows = backupSheet.getDataRange().getValues(); if (bRows.length <= 2) { return ContentService.createTextOutput(JSON.stringify({status: "empty", message: "बैकअप शीट खाली है।"})).setMimeType(ContentService.MimeType.JSON); } var backupTime = bRows[0][1] || "अज्ञात समय"; var bHeaders = bRows[1]; var bItems = []; for (var i = 2; i < bRows.length; i++) { var row = bRows[i]; var obj = {}; for (var j = 0; j < bHeaders.length; j++) { obj[bHeaders[j]] = row[j]; } bItems.push(obj); } return ContentService.createTextOutput(JSON.stringify({status: "ok", items: bItems, timestamp: backupTime})).setMimeType(ContentService.MimeType.JSON); } // सामान्य लोड पर 'Ingredients' शीट से डेटा लाएगा var sheet = ss.getSheetByName("Ingredients"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({status: "empty"})).setMimeType(ContentService.MimeType.JSON); } var rows = sheet.getDataRange().getValues(); if (rows.length <= 1) { return ContentService.createTextOutput(JSON.stringify({status: "empty"})).setMimeType(ContentService.MimeType.JSON); } var headers = rows[0]; var items = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; var obj = {}; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = row[j]; } items.push(obj); } return ContentService.createTextOutput(JSON.stringify({status: "ok", items: items})).setMimeType(ContentService.MimeType.JSON); }
